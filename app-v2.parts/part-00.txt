  const a = vector(start);
  const b = vector(end);
  const direction = b.clone().sub(a);
  object.position.copy(a.clone().add(b).multiplyScalar(0.5));
  object.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), direction.clone().norm
