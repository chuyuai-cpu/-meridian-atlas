function selectChannel(code, resetProgress = false) {
  const channel = channelByCode.get(code);
  if (!channel) return;
  state.channelCode = code;
  if (channel.points.length) state.pointCode = channel.points.find((point) => point.code === state.pointCode)?.code || channel.points[0].code;
  else state.pointCode = null;
  if (resetProgress) state.progress = 0;
  cameraTarget.set(0, 0.05, 0);
  renderUI();
  saveState();
}
function selectPoint(code, side = 'left') {
  const point = pointByCode.get(code);
  if (!point) return;
  state.pointCode = code;
  state.channelCode = point.channelCode;
  state.selectedSide = side;
  state.view = 'point';
  renderUI();
  const channel = activeChannel();
  const position = side === 'right' && channel.bilateral ? mirror(point.position) : point.position;
  cameraTarget.set(...position);
  cameraTarget.y = clamp(cameraTarget.y, -1.6, 1.8);
  state.distance = 5.6;
  showPointToast(point);
  saveState();
}
function showPointToast(point) {
  const toast = $('#pointToast');
  const crossing = point.crossing.length ? ` · 交会：${point.crossing.join('、')}` : '';
  toast.innerHTML = `<strong>${point.code} ${point.name}</strong><span>${point.categories.join(' / ')}${point.element ? ` · 五行${point.element}` : ''}${crossing}</span>`;
  toast.classList.add('show');
  clearTimeout(showPointToast.timer);
  showPointToast.timer = setTimeout(() => toast.classList.remove('show'), 3800);
}

function raycast(event) {
  const rect = canvas.getBoundingClientRect();
  pointerNdc.x = ((event.clientX - rect.left) / rect.width) * 2 - 1;
  pointerNdc.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;
  raycaster.setFromCamera(pointerNdc, camera);
  return raycaster.intersectObjects([...pointObjects, ...[...organObjects.values()]], true).find((hit) => hit.object.userData?.type);
}
