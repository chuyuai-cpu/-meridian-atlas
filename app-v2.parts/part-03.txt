function updateSelectedHalo() {
  if (selectedPointHalo) highlightGroup.remove(selectedPointHalo);
  const point = activePoint();
  if (!point) return;
  const channel = activeChannel();
  const position = state.selectedSide === 'right' && channel.bilateral ? mirror(point.position) : point.position;
  selectedPointHalo = new THREE.Mesh(
    new THREE.RingGeometry(0.085, 0.105, 32),
    new THREE.MeshBasicMaterial({ color: pointColor(point), transparent: true, opacity: 0.82, side: THREE.DoubleSide, depthWrite: false, blending: THREE.AdditiveBlending })
  );
  selectedPointHalo.position.set(...position);
  selectedPointHalo.lookAt(camera.position);
  selectedPointHalo.renderOrder = 11;
  highlightGroup.add(selectedPointHalo);
}

function updateChannelPresentation() {
  const selected = activeChannel();
  channelObjects.forEach(({ channel, group, internal, entries }) => {
    const isSelected = channel.code === selected.code;
    group.visible = state.layers.meridians && (state.showAllChannels || isSelected);
    internal.visible = state.layers.internal && (state.showAllChannels || isSelected);
    entries.forEach(({ mesh, material }) => {
      material.opacity = isSelected ? 0.94 : (channel.polarity === 'yang' ? 0.24 : 0.13);
      mesh.scale.setScalar(isSelected ? 1.7 : 1);
    });
  });
  pointObjects.forEach((mesh) => {
    const isSelected = mesh.userData.channel.code === selected.code;
    mesh.visible = state.layers.points && (isSelected || (state.showAllChannels && state.view === 'point'));
    mesh.material.opacity = isSelected ? 0.98 : 0.30;
    mesh.scale.setScalar(isSelected ? 1.15 : 0.72);
  });
  traceMaterial.color.set(selected.color);
  particles.forEach((particle) => particle.material.color.set(selected.color));
  updateSelectedHalo();
}

function updateLayers() {
  skinGroup.visible = state.layers.skin;
  skeletonGroup.visible = state.layers.skeleton;
  organGroup.visible = state.layers.organs;
  channelGroup.visible = state.layers.meridians;
  pointGroup.visible = state.layers.points;
  internalGroup.visible = state.layers.internal;
  skinMaterial.opacity = state.skinOpacity;
  skinDarkMaterial.opacity = state.skinOpacity;
  skinMaterial.depthWrite = state.skinOpacity > 0.55;
  skinDarkMaterial.depthWrite = state.skinOpacity > 0.55;
  organObjects.forEach((group, id) => {
    group.visible = state.layers.organs && state.organVisibility[id] !== false;
  });
  updateChannelPresentation();
}

function focusView(view) {
  if (view === 'front') { state.yaw = 0; state.pitch = 0.02; }
  if (view === 'back') { state.yaw = Math.PI; state.pitch = 0.02; }
  if (view === 'left') { state.yaw = -Math.PI / 2; state.pitch = 0.02; }
  if (view === 'right') { state.yaw = Math.PI / 2; state.pitch = 0.02; }
  state.distance = 8.3;
  saveState();
}
function focusOrgan(id) {
  const organ = DATA.organs.find((item) => item.id === id);
  if (!organ) return;
  cameraTarget.set(...organ.position);
  state.distance = 4.8;
  state.yaw = 0;
  state.pitch = 0.08;
}
